# Phase 2.7 PR A — Translated Interface Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every string in the frontend comes from a per-language catalogue, the user's language is stored on their row and followed on every device, and backend errors render in that language — with no change to retrieval or answers.

**Architecture:** The backend gains a `Language` catalogue, a nullable `users.language` column behind `GET`/`PUT /me/language`, a `language` field on login/refresh responses, and an additive `fieldTypes` map on `422` bodies. The frontend adopts `next-intl` without locale-prefixed routes: the locale is resolved per request from an httpOnly `askrepo_locale` cookie (which mirrors the user row and is written only by the existing cookie writers), then `Accept-Language`, then `en`. Errors are translated by `ErrorCode` and Pydantic error type at render time; `eslint-plugin-i18next` enforces that no JSX literal survives.

**Tech Stack:** FastAPI, SQLAlchemy, Alembic, pytest (backend); Next.js 16.3, React 19, `next-intl` 4.14.9, `eslint-plugin-i18next` 6.1.5, Vitest (frontend).

**Spec:** `docs/superpowers/specs/2026-10-17-phase-2.7-multi-language-design.md` — §0.1, §1 (users.language only), §2, §8 "PR A". Projects' `content_language`, `PATCH /projects/{id}`, eval language, prompts, refusals and email are PR B / PR C and are **not** in this plan.

## Global Constraints

- Languages, exactly: `en`, `id`, `ja`, `ms`, `zh-HK`, `zh-TW`, `ko` (BCP-47). Values are a wire contract: add, never rename.
- `users.language` is `String(8)`, nullable; `NULL` means "never chose". An administrator never writes it.
- No locale-prefixed routes. No URL changes anywhere.
- `askrepo_locale` is httpOnly, `Path=/`, set through `cookieOptions(...)` in `frontend/lib/auth/cookies.ts` and nowhere else.
- A `null` language on a login/refresh response leaves `askrepo_locale` untouched; only `PUT /me/language` with `null` clears it. (Spec amendment, Task 20.)
- No `zod`, no `react-hook-form` (`.claude/rules/forms.md` §4). Client-side checks stay hand-written; only their messages move to the catalogue.
- the English catalogue (`messages/en/`) is the source of keys. Every other message file has exactly its keys and ICU placeholders. Each locale is a directory of per-namespace files (`messages/<locale>/<namespace>.json` plus `index.ts`); non-English locales carry `_meta.json` = `{ "reviewed": false }`.
- No downloaded font. CJK fallback is a system font stack in `globals.css`.
- Never a `dark:` colour utility, never a palette utility (`CLAUDE.md` → Frontend).
- Backend: every function annotated, docstring on every public unit, `ruff` clean (`.claude/rules/clean-code.md`).
- Commit after each task with a Conventional Commit message ending in `Co-Authored-By: Claude Opus 5.5 (1M context) <noreply@anthropic.com>`. Work on branch `feat/phase-2.7a-interface-language`, created from `main` after the spec branch merges.
- Read `frontend/AGENTS.md`: this Next.js has breaking changes; check `frontend/node_modules/next/dist/docs/` before relying on any Next API not already used in the tree.

```python
    async def language(self, user: AuthenticatedUser) -> LanguageSetting:
        """The caller's stored interface language, `None` if they never chose one."""
        row = await self._own_row(user)
        return LanguageSetting(language=row.language)

    async def update_language(
        self, user: AuthenticatedUser, payload: LanguageSetting
    ) -> LanguageSetting:
        """Replace the caller's language, and audit it after the commit if it moved."""
        row = await self._own_row(user)
        before = row.language
        after = _stored(payload.language)
        row.language = after
        await self.session.commit()

        if before != after:
            await self._recorder.record(
                AuditEntry(
                    event_type=AuditEventType.USER_LANGUAGE_UPDATED,
                    actor_user_id=user.id,
                    actor_email=user.email,
                    target_type="user",
                    target_id=user.id,
                    target_label=user.email,
                    ip_address=self._client_ip,
                    changed={"language": (before, after)},
                )
            )
        return LanguageSetting(language=payload.language)
```

```ts
/**
 * The languages AskRepo ships. Mirrors `backend/app/core/language.py`, and
 * `backend/tests/test_language_catalogue.py` fails when the two differ.
 */
export const LANGUAGES = ["en", "id", "ja", "ms", "zh-HK", "zh-TW", "ko"] as const;

export type Language = (typeof LANGUAGES)[number];

export const DEFAULT_LANGUAGE: Language = "en";

/** Each language in its own name, so a user finds theirs without reading the current one. */
export const ENDONYMS: Record<Language, string> = {
  en: "English",
  id: "Bahasa Indonesia",
  ja: "日本語",
  ms: "Bahasa Melayu",
  "zh-HK": "繁體中文（香港）",
  "zh-TW": "繁體中文（台灣）",
  ko: "한국어",
};

export function isLanguage(value: unknown): value is Language {
  return typeof value === "string" && (LANGUAGES as readonly string[]).includes(value);
}

/**
 * Chinese tags that are not an exact match. Simplified (`zh`, `zh-CN`, `zh-Hans`, `zh-SG`)
 * is deliberately absent: showing a Simplified reader a Traditional catalogue is worse
 * than English.
 */
const CHINESE_ALIASES: Record<string, Language> = {
  "zh-mo": "zh-HK",
  "zh-hant": "zh-TW",
  "zh-hant-hk": "zh-HK",
  "zh-hant-mo": "zh-HK",
  "zh-hant-tw": "zh-TW",
};

function matchTag(tag: string): Language | undefined {
  const lower = tag.toLowerCase();
  const exact = LANGUAGES.find((code) => code.toLowerCase() === lower);
  if (exact) return exact;
  if (lower in CHINESE_ALIASES) return CHINESE_ALIASES[lower];
  const base = lower.split("-")[0];
  if (base === "zh") return undefined;
  return LANGUAGES.find((code) => code === base);
}

/** Accept-Language tags, highest q first; ties keep header order. */
function rankedTags(header: string): string[] {
  return header
    .split(",")
    .map((part, index) => {
      const [tag, ...params] = part.trim().split(";");
      const q = params.find((p) => p.trim().startsWith("q="));
      return { tag: tag.trim(), q: q ? Number(q.trim().slice(2)) : 1, index };
    })
    .filter((entry) => entry.tag && entry.tag !== "*" && !Number.isNaN(entry.q))
    .sort((a, b) => b.q - a.q || a.index - b.index)
    .map((entry) => entry.tag);
}

/**
 * The locale for one request: the `askrepo_locale` cookie (which mirrors the user
 * row), then `Accept-Language`, then English. Spec §2.2.
 */
export function resolveLocale(
  cookie: string | undefined,
  acceptLanguage: string | null,
): Language {
  if (isLanguage(cookie)) return cookie;
  for (const tag of rankedTags(acceptLanguage ?? "")) {
    const match = matchTag(tag);
    if (match) return match;
  }
  return DEFAULT_LANGUAGE;
}
```

```ts
import type { NextResponse } from "next/server";

import { LOCALE_COOKIE, SESSION_MAX_AGE_SECONDS, cookieOptions } from "@/lib/auth/cookies";
import { isLanguage } from "@/lib/i18n/languages";

/**
 * Mirror a stored language onto the locale cookie. `null` and anything unshipped
 * leave the cookie as it is: a user with no stored language keeps whatever this
 * browser chose, rather than having it wiped at every login (plan Global Constraints).
 */
export function applyLocale(response: NextResponse, language: unknown): void {
  if (isLanguage(language)) {
    response.cookies.set(LOCALE_COOKIE, language, cookieOptions(SESSION_MAX_AGE_SECONDS));
  }
}

/** Forget this browser's language, so `Accept-Language` decides again. */
export function clearLocale(response: NextResponse): void {
  response.cookies.set(LOCALE_COOKIE, "", cookieOptions(0));
}
```

```ts
import { NextResponse } from "next/server";

import { applyLocale } from "@/lib/auth/locale-cookie";
import { isLanguage } from "@/lib/i18n/languages";

/**
 * Set this browser's language before anyone has signed in. There is no account to
 * store it on yet, so nothing reaches the backend; logging in to an account that has
 * a stored language replaces it (spec §2.2). A cookie writer under `/api/auth/*`,
 * per `.claude/rules/frontend-bff.md`.
 */
export async function POST(request: Request): Promise<Response> {
  let language: unknown;
  try {
    language = ((await request.json()) as { language?: unknown })?.language;
  } catch {
    language = undefined;
  }

  if (!isLanguage(language)) {
    return NextResponse.json(
      {
        detail: {
          code: "VALIDATION_ERROR",
          message: "Unsupported language.",
          fields: { language: "Unsupported language." },
          fieldTypes: { language: "enum" },
        },
      },
      { status: 422 },
    );
  }

  const response = new NextResponse(null, { status: 204 });
  applyLocale(response, language);
  return response;
}
```

```ts
"use client";

import { useTranslations } from "next-intl";

import { isApiError } from "@/lib/api/errors";

/**
 * An error as the user should read it. Translated by `ErrorCode` -- a wire contract
 * that never changes -- never by the backend's English `message`, which is the
 * fallback when a code has no entry (spec §2.4).
 */
export function useErrorText(): (error: unknown) => string {
  const t = useTranslations("errors");
  return (error) => {
    if (!isApiError(error)) return t("generic");
    if (error.status === 0) return t("network");
    return t.has(error.code) ? t(error.code) : error.message;
  };
}

/**
 * One field's error. Translated by Pydantic's error type when the backend sent one;
 * otherwise the backend's own text, which is how a service-level rule such as the
 * password policy still reaches the user (spec §2.5).
 */
export function useFieldErrorText(): (error: unknown, name: string) => string | undefined {
  const t = useTranslations("forms.validation");
  return (error, name) => {
    if (!isApiError(error)) return undefined;
    const message = error.fieldErrors[name];
    if (message === undefined) return undefined;
    const type = error.fieldTypes[name];
    return type && t.has(type) ? t(type) : message;
  };
}
```

```ts
function relativeFormat(locale: string): Intl.RelativeTimeFormat {
  let formatter = formatters.get(locale);
  if (!formatter) {
    formatter = new Intl.RelativeTimeFormat(locale, { numeric: "auto" });
    formatters.set(locale, formatter);

  return formatter;

export function formatRelative(iso: string, locale: string, now: number = Date.now()): string {
  const elapsed = new Date(iso).getTime() - now;

  for (const [unit, size] of UNITS) {
```

These seven tasks are one procedure applied to seven areas. Each is a separate task and commit so a reviewer can reject one area without the others. The procedure is spelled out once here and is **the same for every area**; each task below names its files, its namespace, and anything specific to it.

**Execution order and file ownership.** Task 12 runs first and alone: it changes shared helpers (`lib/nav.ts`, `lib/status.ts`, `lib/password-rules.ts`) whose callers live in every area. Tasks 13–17 then run **in parallel**, each in its own git worktree, and Task 18 runs last, after they are merged. Parallel tasks must never touch the same file, so in Tasks 13–17:

- **Write only your own namespace files**, `messages/en/<your namespace>.json`. `common.json` is read-only (seeded in Task 5): a shared word it lacks goes in your own namespace. `index.ts` is never edited — every namespace file already exists.
- **Edit only the files your task lists, plus the hooks named after your area** (`hooks/use-projects.ts`, `hooks/use-project-mutations.ts`, `hooks/use-members.ts`, `hooks/use-member-mutations.ts`, `hooks/use-indexed-paths.ts` belong to Task 13; `use-ask-stream`, `use-conversations`, `use-feedback` to 14; `use-checklist*`, `use-mock-data*` to 15; `use-eval` to 16; `use-users`, `use-user-mutations`, `use-roles`, `use-role-mutations`, `use-role-options`, `use-audit-events`, `use-notifications` to 17). - **Do not edit `eslint.config.mjs`.** Step 8 below applies to Task 12 only; Tasks 13–17 verify with the command in Task 18's sweep, restricted to their own directories, and Task 19 turns the rule on for the whole tree.

**The procedure (for every file in the area):**

1. **Client Components** (`"use client"`): `const t = useTranslations("<namespace>");` at the top of the component; every JSX text node and every `aria-label`/`placeholder`/`title`/`alt` string becomes `t("key")`. **Server Components** (no `"use client"`, often `page.tsx`): `const t = await getTranslations("<namespace>");` from `next-intl/server`, and the component must be `async`. Never call `useTranslations` in a Server Component.
2. **Keys** are camelCase, grouped by component: `projects.createDialog.title`, `projects.table.columns.status`. Put each string into `messages/en/<namespace>.json` verbatim — the English the user sees must not change in this task.
3. **Interpolation and plurals use ICU**, never string concatenation:
   - `` `${count} pairs` `` → `t("pairCount", { count })` with `"pairCount": "{count, plural, one {# pair} other {# pairs}}"`.
   - `Deleted {name}` → `t("deleted", { name })` with `"deleted": "Deleted {name}"`.
   - Text with inline markup (`Read <Link>the docs</Link>`) → `t.rich("key", { link: (chunks) => <Link href="…">{chunks}</Link> })` with `"key": "Read <link>the docs</link>"`.
   - **No ICU `select`.** A branch per value (`{role, select, owner {Owner} …}`) is one key per value instead (`roles.owner`, chosen in code). The parity test (Task 19) reads `{word}` as a placeholder, so a translated `select` branch would fail it — and separate keys are easier to translate anyway.
4. **Toasts** (`toast.success("…")`, `toast.error("…")`) and **`confirm`/dialog copy** passed as props move too, even though lint does not catch non-JSX strings. After the area is done, `grep -rnE 'toast\.(success|error|info|warning)\("' <area dirs>` must print nothing.
5. **Errors:** every `fieldError(error, "x")` becomes `fieldErrorText(error, "x")` with `const fieldErrorText = useFieldErrorText();`; every direct render of `error.message` becomes `errorText(error)` with `const errorText = useErrorText();` (Task 8).

For a label map that a pure function returns (`lib/status.ts`-style), return a message key from the function and translate where it renders; update its unit test to assert the key.

```ts
import { join } from "node:path";

import { describe, expect, it } from "vitest";

import { LANGUAGES } from "@/lib/i18n/languages";

type Tree = { [key: string]: string | Tree };

const ROOT = join(__dirname, "../../messages");

/** One locale's namespaces, keyed by file name; `_meta.json` is not a namespace. */
function load(locale: string): Tree {
  const tree: Tree = {};
  for (const file of readdirSync(join(ROOT, locale))) {
    if (!file.endsWith(".json") || file === "_meta.json") continue;
    tree[file.slice(0, -".json".length)] = JSON.parse(
      readFileSync(join(ROOT, locale, file), "utf8"),
    );
  }
  return tree;
}

function meta(locale: string): unknown {
  return JSON.parse(readFileSync(join(ROOT, locale, "_meta.json"), "utf8"));
}

function leaves(tree: Tree, prefix = ""): Map<string, string> {
  const out = new Map<string, string>();
  for (const [key, value] of Object.entries(tree)) {
    if (key === "_meta") continue;
    const path = prefix ? `${prefix}.${key}` : key;
    if (typeof value === "string") out.set(path, value);
    else for (const [k, v] of leaves(value, path)) out.set(k, v);
  }
  return out;
}

/** ICU argument names in a message: `{count}`, `{count, plural, …}`, `<link>`. */
function placeholders(message: string): string[] {
  const args = [...message.matchAll(/\{\s*([A-Za-z0-9_]+)\s*[,}]/g)].map((m) => m[1]);
  const tags = [...message.matchAll(/<([A-Za-z0-9_]+)>/g)].map((m) => `<${m[1]}>`);
  return [...new Set([...args, ...tags])].sort();
}

const english = leaves(load("en"));

  });

  it("keeps every placeholder and tag", () => {
    for (const [key, message] of english) {
      expect(placeholders(translated.get(key) ?? ""), key).toEqual(placeholders(message));
    }
  });

  it("declares whether a native speaker has reviewed it", () => {

  });
```
